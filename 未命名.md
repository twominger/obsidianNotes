### 安装插件
```shell
# show plugins; | grep group_replication
install plugin group_replication soname 'group_replication.so';
```
### 配置账号
```shell
set session sql_log_bin=0;
CREATE USER 'sportsmate'@'172.16.1.%' IDENTIFIED BY 'sportsmate';
GRANT REPLICATION SLAVE ON *.* TO 'sportsmate'@'172.16.1.%';
flush privileges;
set session sql_log_bin=1;

# 配置MGR服务通道
change master to master_user='sportsmate',master_password='sportsmate'  for channel 'group_replication_recovery';
```
### master 节点启动引导，进入 mysql 服务端
```shell
set global group_replication_bootstrap_group=on;
start group_replication;
set global group_replication_bootstrap_group=OFF;
```
### slave 节点进入 mysql 服务端
```shell
set global group_replication_allow_local_disjoint_gtids_join=ON;
start group_replication;
select * from performance_schema.replication_group_members;
```

```shell
cat >/etc/keepalived/check_apiserver.sh <<EOF
#!/bin/bash

err=0
for k in $(seq 1 3)
do
    check_code=$(pgrep mysql)
    if [[ $check_code == "" ]]; then
        err=$(expr $err + 1)
        sleep 1
        continue
    else
        err=0
        break
    fi
done

if [[ $err != "0" ]]; then
    echo "systemctl stop keepalived"
    /usr/bin/systemctl stop keepalived
    exit 1
else
    exit 0
fi
EOF

chmod +x /etc/keepalived/check_apiserver.sh

sed -i '/#.*track_script {/ s/^#//' /etc/keepalived/keepalived.conf
sed -i '/#.*chk_apiserver/ s/^#//' /etc/keepalived/keepalived.conf
sed -i '/#.*}/ s/^#//' /etc/keepalived/keepalived.conf

systemctl enable keepalived
systemctl restart keepalived
```

```shell
# 创建命名空间
---
apiVersion: v1
kind: Namespace
metadata:
  name: sportsmate

# 使用 deployment 创建资源
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: sportsmate-deployment  # Deployment 名称
  namespace: sportsmate        # 指定命名空间
  labels:
    app: sportsmate            # 应用标签
spec:
  replicas: 3                  # 副本数量
  selector:
    matchLabels:
      app: sportsmate          # 选择器标签，必须与 Pod 模板中的标签匹配
  template:
    metadata:
      labels:
        app: sportsmate        # Pod 标签
    spec:
      containers:
      - name: sportsmate-container  # 容器名称
        image: crpi-on4n8clbhol74dg8.cn-hangzhou.personal.cr.aliyuncs.com/superming/sportsmate:v1  # 镜像地址
        ports:
        - containerPort: 8080  # 容器暴露的端口（Spring Boot 默认端口）
        env:
        - name: SPRING_PROFILES_ACTIVE
          value: "prod"        # 指定 Spring Boot 的激活配置文件
        resources:
          requests:
            memory: "512Mi"    # 容器请求的内存
            cpu: "500m"        # 容器请求的 CPU
          limits:
            memory: "1Gi"      # 容器内存限制
            cpu: "1000m"       # 容器 CPU 限制
        livenessProbe:         # 存活探针
          httpGet:
            path: /actuator/health  # 健康检查路径
            port: 8080
          initialDelaySeconds: 30   # 延迟启动探针的时间
          periodSeconds: 10         # 探针检查间隔时间
        readinessProbe:        # 就绪探针
          httpGet:
            path: /actuator/health  # 健康检查路径
            port: 8080
          initialDelaySeconds: 30   # 延迟启动探针的时间
          periodSeconds: 10         # 探针检查间隔时间
      restartPolicy: Always    # 重启策略

# 定义资源服务暴露方式
---
apiVersion: v1
kind: Service
metadata:
  name: sportsmate-service  # Service 名称
  namespace: sportsmate     # 指定命名空间
spec:
  selector:
    app: sportsmate         # 选择器标签，必须与 Pod 模板中的标签匹配
  ports:
  - protocol: TCP
    port: 80                # Service 端口
    targetPort: 8080        # 容器端口
  type: LoadBalancer        # 使用 LoadBalancer 类型暴露服务


# 创建 Horizontal Pod Autoscaler (HPA)

  - type: Resource
```
